\begin{definition}
	A Petrinet $\Omega$ is a directed graph $(V, E)$ with vertices $V=V_X \cup
		V_Z$ partitioned into sets $V_X$ of state vertices and $V_Z$ of transition
	vertices.
\end{definition}

\begin{definition}
	Let $E(v) = E^{in}(v) \cup E^{out}(v)$, and for each $v\in V$ denote edges of
	the form   $(v, v') \in E^{out}(v)$ or  $(v', v) \in E^{in}(v)$ where $v$ is
	the respective source or target.
\end{definition}

\begin{example}
	The SIR model defines $\Omega$ by:
	\begin{eqnarray*}
		V_X &=& \{v_{S}, v_{I}, v_{R}\}\\
		V_Z &=& \{ v_{inf}, v_{rec}\}\\
		E &=& \{ (v_{S}, v_{inf}),(v_{I}, v_{inf}), (v_{inf}, v_{I}), (v_{inf}, v_{I}),  (v_{I}, v_{rec}), (v_{rec}, v_{R})\}
	\end{eqnarray*}
        This is also depicted in Figure \ref{fig:sir_model}.
\end{example}

\begin{definition}
	The ODE semantics $\Theta$ of the Petrinet $\Omega$ defines a tuple $(P, D, X,
		Z, {\cal I}, {\cal P}, {\cal X}, {\cal Z}, {\cal R})$ where
	\begin{itemize}
		\item $P$ is a set of parameters;
		\item $D$ is a set of discrete state variables, and for each $d \in
		D$, $dom(d)$ is a finite set of values;
		\item $X$ is a set of continuous state variables;
		\item $Z$ is a set of transitions;
		\item ${\cal P}: P \rightarrow \reals \cup (\reals \times \reals)$ assigns
		      parameters to a real number, or a pair of real numbers defining an
		      interval;
		\item ${\cal X}: V_X  \rightarrow X$ assigns a
		continuous variable to each state
		vertex;
		\item ${\cal D}: V_X \rightarrow 2^{St}$ maps a set of discrete states
		to each $v \in V_X$, where $St = dom(d_1) \times \ldots
		\times dom(d_{|D|})$ is the set of all discrete states.
		\item ${\cal I}: X \rightarrow \reals$ assigns the initial value of
		      continuous state variables to a real number;
		\item ${\cal Z}: Z \rightarrow V_z$ assigns transitions to transition
		      vertices; and
		\item ${\cal R}: {\bf P} \times V_X \times Z \rightarrow \reals$
		      defines the rate of each transition  $z \in Z$ in terms of the set of
		      parameter vectors ${\bf P}$ and state variable vectors ${\bf X}$.
	\end{itemize}

 The semantics associates a pair $({\cal X}(v),{\cal D}(v)) = (x, S)$ with each vertex $v \in V_X$ where $x
\in X$ and $S \subseteq St$.  We assume that the mapping of continuous variables
and discrete states to vertices is complete, so that for each $x \in X$ there is a
$v \in V_X$ where ${\cal X}(v) = x$.  The set of vertices $V(x)=\{v \in V_X | {\cal
X}(v) = x\}$ associated with $x$ also partition the discrete states, so
that $St = \bigcup_{v\in V(x)}{\cal D}(v)$ and for each $v_1, v_2 \in V(x)$,
${\cal D}(v_1)\cap{\cal D}(v_2) = \emptyset$.  

Let ${\bf x}$ denote the vector of pairs $(x, S)$ mapped to vertices in $V_X$,
and let
$x_S$ denote an element of ${\bf x}$. The elements of the Petrinet $\Omega$ and semantics $\Theta$ define the
partial derivative $\frac{d {\bf x}}{dt}$, so that for each $x_S$ which maps to
a vertex $v$:

	\begin{equation}\label{eqn:flow}
		\frac{dx_S}{dt} = \sum_{(v_z, v) \in E} {\cal R}({\bf p}, {\bf x}, z) - \sum_{(v, v_z)\in E}  {\cal R}({\bf p}, {\bf x}, z)
	\end{equation}
	
	\noindent We denote
	by $\nabla_{\Omega, \Theta}({\bf p}, {\bf x}, t)$, the gradient comprised of components defined in
	Equation \ref{eqn:flow}.
\end{definition}


\begin{definition}
	A balanced-flow Petrinet preserves the flow through each transition by requiring that $|E^{in}(v_z)| = |E^{out}(v_z)|$ for each $v_z \in V_Z$.
\end{definition}

\begin{definition}
Each balanced-flow Petrinet transition $z$ defines a set of state transitions 
 $st(z) = \{((x, S), z,
 (x', S')) | (v, z) \in E^{in}(z), ({\cal X}(v),{\cal D}(v)) = (x, S),  (z, v') \in E^{out}(z), ({\cal X}(v'),{\cal D}(v')) = (x', S')\}$ that match each input state with exactly one output state, and
 vice versa.
\end{definition}

\begin{definition}
	An independent-transition Petrinet requires that for each transition, there
	is exactly one $(x, S), z,
	(x', S') \in st(z)$ where $(x, S)\not= (x', S')$.
\end{definition}

A balanced-flow, independent-transition Petrinet permits only a single state
change with each transition.  The remaining state transitions define the variables that impact this state change, and no transition
allows more than one state change.  These definitions become important for
defining Petrinet stratification, as described in the following section.

semantics because the graph vertices and the semantic elements are one to one.

Using the partial derivatives defined by the Petrinet graph and semantics, we
can define the state vector at given time $t+dt$ with the forward Euler method
as:

\begin{eqnarray*}
	\frac{d {\bf x}}{dt} &=& \nabla_{\Omega, \Theta}({\bf p},{\bf x}, t)\\
	\frac{{\bf x}(t+dt)-{\bf x}(t)}{dt} &=& \nabla_{\Omega, \Theta}({\bf p},{\bf x},
	t)\\
	{\bf x}(t+dt)&=& \nabla_{\Omega, \Theta}({\bf p},{\bf x}, t)dt+ {\bf x}(t)
\end{eqnarray*}
